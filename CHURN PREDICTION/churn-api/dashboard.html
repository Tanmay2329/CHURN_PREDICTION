Integration Examples
import requests

# Single prediction
payload = {
    "tenure": 8, "monthly_charges": 75.50, "total_charges": 604.0,
    "num_products": 2, "support_calls": 4, "days_since_login": 30,
    "has_tech_support": 0, "plan": "standard",
    "contract_type": "month-to-month", "payment_method": "e-check"
}

r = requests.post("http://localhost:8000/predict", json=payload)
result = r.json()

print(result["churn_probability"])  # 0.3241
print(result["risk_tier"])         # "medium"

# Batch prediction
batch_r = requests.post("http://localhost:8000/predict/batch",
    json={"customers": [payload, payload]})
print(batch_r.json()["high_risk_count"])

const predictChurn = async (customer) => {
  const res = await fetch("http://localhost:8000/predict", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(customer)
  });
  return res.json();
};

const result = await predictChurn({
  tenure: 8, monthly_charges: 75.50, total_charges: 604.0,
  num_products: 2, support_calls: 4, days_since_login: 30,
  has_tech_support: 0, plan: "standard",
  contract_type: "month-to-month", payment_method: "e-check"
});

console.log(`Risk: ${result.risk_tier} · ${(result.churn_probability*100).toFixed(1)}%`);
