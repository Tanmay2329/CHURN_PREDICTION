/predict?threshold=0.5
Single customer churn score
import requests payload = { "tenure": 8, "monthly_charges": 75.50, "total_charges": 604.0, "num_products": 2, "support_calls": 4, "days_since_login": 30, "has_tech_support": 0, "plan": "standard", "contract_type": "month-to-month", "payment_method": "e-check" } r = requests.post("http://localhost:8000/predict?threshold=0.3", json=payload) print(r.json()) # {"churn_probability":0.3241,"churn_prediction":true,"risk_tier":"medium","threshold_used":0.3}
curl -X POST "http://localhost:8000/predict?threshold=0.3" \ -H "Content-Type: application/json" \ -d '{"tenure":8,"monthly_charges":75.5,"total_charges":604, "num_products":2,"support_calls":4,"days_since_login":30, "has_tech_support":0,"plan":"standard", "contract_type":"month-to-month","payment_method":"e-check"}'
const res = await fetch("http://localhost:8000/predict?threshold=0.3", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ tenure: 8, monthly_charges: 75.5 /* ... */ }) }); const data = await res.json(); console.log(data.risk_tier); // "medium"
